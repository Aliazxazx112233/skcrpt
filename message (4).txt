local function CleanupAll()
    if CharacterChatConnection then CharacterChatConnection:Disconnect() end
    if TextPropertyConnection then TextPropertyConnection:Disconnect() end
    if FollowConnection then FollowConnection:Disconnect() end
    if JumpConnection then JumpConnection:Disconnect() end
    if ToolConnection then ToolConnection:Disconnect() end
    if ToolCharacterConnection then ToolCharacterConnection:Disconnect() end
    if ToolBackpackConnection then ToolBackpackConnection:Disconnect() end

    for _, con in ipairs(GlobalChatConnections) do
        pcall(function()
            con:Disconnect()
        end)
    end

    ScreenGui:Destroy()
end

CloseBtn.MouseButton1Click:Connect(CleanupAll)

local function IsCommand(text)
    if not text then return false end

    local clean = text:match("^%s*(.-)%s*$")
    local firstChar = clean:sub(1, 1)

    return firstChar == "/" or firstChar == ";" or firstChar == "!" or firstChar == "."
end

-- مهم:
-- يمنع أي TextLabel/TextBox تابع لـ Tool من اعتباره شات
local function IsInsideTool(instance)
    if not instance then
        return false
    end

    local current = instance

    while current and current ~= workspace do
        if current:IsA("Tool") then
            return true
        end

        current = current.Parent
    end

    return false
end

local function CleanChatText(text)
    if not text then
        return ""
    end

    return tostring(text)
        :gsub("<[^>]->", "")
        :gsub("<[^>]*>", "")
        :gsub("^%s+", "")
        :gsub("%s+$", "")
end

local function GlobalChatListener(player)
    if player == LocalPlayer then
        return
    end

    local function MonitorTextLabel(label)
        if not label or not label.Parent then
            return
        end

        -- تجاهل أي نص داخل Tool بالكامل
        if IsInsideTool(label) then
            return
        end

        local cleanMsg = CleanChatText(label.Text)

        if cleanMsg ~= "" and IsCommand(cleanMsg) then
            AddCommandToList(player, cleanMsg)
        end
    end

    local function HookCharacter(char)
        if not char then
            return
        end

        for _, descendant in ipairs(char:GetDescendants()) do
            if descendant:IsA("TextLabel") or descendant:IsA("TextBox") then
                MonitorTextLabel(descendant)

                local propertyConnection = descendant:GetPropertyChangedSignal("Text"):Connect(function()
                    MonitorTextLabel(descendant)
                end)

                table.insert(GlobalChatConnections, propertyConnection)
            end
        end

        local c2 = char.DescendantAdded:Connect(function(descendant)
            if descendant:IsA("TextLabel") or descendant:IsA("TextBox") then
                MonitorTextLabel(descendant)

                local propertyConnection = descendant:GetPropertyChangedSignal("Text"):Connect(function()
                    MonitorTextLabel(descendant)
                end)

                table.insert(GlobalChatConnections, propertyConnection)
            end
        end)

        table.insert(GlobalChatConnections, c2)
    end

    local c1 = player.CharacterAdded:Connect(function(char)
        HookCharacter(char)
    end)

    table.insert(GlobalChatConnections, c1)

    if player.Character then
        HookCharacter(player.Character)
    end
end

for _, p in ipairs(Players:GetPlayers()) do
    GlobalChatListener(p)
end

table.insert(GlobalChatConnections, Players.PlayerAdded:Connect(GlobalChatListener))

-- حماية النص من رموز Lua pattern
local function EscapePattern(text)
    return text:gsub("([%(%)%.%%%+%-%*%?%[%]%^%$])", "%%%1")
end

-- عكس النص مع المحافظة على علامات الترقيم
local function FilterAndReplaceWords(text)
    if not text or text == "" then
        return text
    end

    local modifiedText = text

    -- العبارات المتعددة الكلمات أولاً
    -- حتى لا تتدخل عمليات استبدال الكلمات المفردة معها
    local phraseReplacements = {
        ["تاج راسك"] = "علي تاج راسك",
        ["من انت"] = "من علي عمك",
        ["من أنت"] = "من علي عمك",
        ["مين انت"] = "مين علي عمك",
        ["مين أنت"] = "مين علي عمك",
        ["فزت عليك"] = "فزت على نفسك",
        ["جلدتك"] = "جلدت نفسك",
        ["دعستك"] = "دعست نفسك",
        ["قتلتك"] = "قتلت نفسك"
    }

    for phrase, replacement in pairs(phraseReplacements) do
        local escapedPhrase = EscapePattern(phrase)

        modifiedText = modifiedText:gsub(
            "(^)" .. escapedPhrase .. "([%s%p])",
            "%1" .. replacement .. "%2"
        )

        modifiedText = modifiedText:gsub(
            "([%s])" .. escapedPhrase .. "([%s%p])",
            "%1" .. replacement .. "%2"
        )

        modifiedText = modifiedText:gsub(
            "([%s])" .. escapedPhrase .. "$",
            "%1" .. replacement
        )
    end

    -- الكلمات المفردة
    modifiedText = modifiedText:gsub("%S+", function(token)
        local leading = ""
        local trailing = ""
        local core = token

        -- حفظ علامات البداية
        while core ~= "" do
            local first = core:sub(1, 1)

            if first == "("
                or first == ")"
                or first == "["
                or first == "]"
                or first == "{"
                or first == "}"
                or first == "\""
                or first == "'"
                or first == "!"
                or first == "?"
                or first == ","
                or first == "."
                or first == ":"
                or first == ";"
                or first == "؟"
                or first == "،"
                or first == "؛" then

                leading = leading .. first
                core = core:sub(2)
            else
                break
            end
        end

        -- حفظ علامات النهاية
        while core ~= "" do
            local last = core:sub(-1)

            if last == "("
                or last == ")"
                or last == "["
                or last == "]"
                or last == "{"
                or last == "}"
                or last == "\""
                or last == "'"
                or last == "!"
                or last == "?"
                or last == ","
                or last == "."
                or last == ":"
                or last == ";"
                or last == "؟"
                or last == "،"
                or last == "؛" then

                trailing = last .. trailing
                core = core:sub(1, -2)
            else
                break
            end
        end

        local replacement = WordReplacements[core]

        if replacement then
            return leading .. replacement .. trailing
        end

        return token
    end)

    return modifiedText
end

local function GetEquippedTool(character)
    if not character then
        return nil
    end

    for _, child in ipairs(character:GetChildren()) do
        if child:IsA("Tool") then
            return child
        end
    end

    return nil
end

local function FindMatchingTool(backpack, toolName)
    if not backpack or not toolName then
        return nil
    end

    -- بحث مطابق للاسم
    local exact = backpack:FindFirstChild(toolName)

    if exact and exact:IsA("Tool") then
        return exact
    end

    -- احتياط في حال كان هناك اختلاف بسيط
    for _, item in ipairs(backpack:GetChildren()) do
        if item:IsA("Tool") and item.Name == toolName then
            return item
        end
    end

    return nil
end

local function StopToolMatching()
    if ToolConnection then
        ToolConnection:Disconnect()
        ToolConnection = nil
    end

    if ToolCharacterConnection then
        ToolCharacterConnection:Disconnect()
        ToolCharacterConnection = nil
    end

    if ToolBackpackConnection then
        ToolBackpackConnection:Disconnect()
        ToolBackpackConnection = nil
    end
end

local function StartToolMatching()
    StopToolMatching()

    if not TargetPlayer then
        return
    end

    local function SyncTool()
        if not IsToolMatching then
            return
        end

        if not TargetPlayer then
            return
        end

        local targetChar = TargetPlayer.Character
        local myChar = LocalPlayer.Character
        local myBackpack = LocalPlayer:FindFirstChildOfClass("Backpack")

        if not targetChar or not myChar or not myBackpack then
            return
        end

        local targetTool = GetEquippedTool(targetChar)
        local myTool = GetEquippedTool(myChar)
        local myHumanoid = myChar:FindFirstChildOfClass("Humanoid")

        if not myHumanoid then
            return
        end

        -- اللاعب المستهدف ماسك أداة
        if targetTool then
            if myTool and myTool.Name == targetTool.Name then
                return
            end

            local matchingTool = FindMatchingTool(myBackpack, targetTool.Name)

            if matchingTool then
                pcall(function()
                    myHumanoid:EquipTool(matchingTool)
                end)
            end

        -- اللاعب المستهدف لا يمسك أي أداة
        else
            if myTool then
                pcall(function()
                    myHumanoid:UnequipTools()
                end)
            end
        end
    end

    -- أول مزامنة
    SyncTool()

    local function HookTargetCharacter(character)
        if not character then
            return
        end

        if ToolConnection then
            ToolConnection:Disconnect()
            ToolConnection = nil
        end

        -- يلتقط:
        -- Tool يدخل اليد
        -- Tool يخرج من اليد
        -- تبديل الأداة
        ToolConnection = character.ChildAdded:Connect(function(child)
            if child:IsA("Tool") then
                task.defer(SyncTool)
            end
        end)

        local childRemovedConnection = character.ChildRemoved:Connect(function(child)
            if child:IsA("Tool") then
                task.defer(SyncTool)
            end
        end)

        -- نحتفظ به مع بقية الاتصالات
        table.insert(GlobalChatConnections, childRemovedConnection)

        task.defer(SyncTool)
    end

    if TargetPlayer.Character then
        HookTargetCharacter(TargetPlayer.Character)
    end

    -- إذا مات الهدف وظهرت شخصية جديدة
    ToolCharacterConnection = TargetPlayer.CharacterAdded:Connect(function(character)
        task.wait(0.15)

        if IsToolMatching then
            HookTargetCharacter(character)
            SyncTool()
        end
    end)

    -- إذا ظهرت الأداة في حقيبتي بعد ما كانت غير موجودة
    local myBackpack = LocalPlayer:FindFirstChildOfClass("Backpack")

    if myBackpack then
        ToolBackpackConnection = myBackpack.ChildAdded:Connect(function(child)
            if child:IsA("Tool") then
                task.defer(SyncTool)
            end
        end)
    end
end

            local tHum = targetChar:FindFirstChildOfClass("Humanoid")

                and myChar:FindFirstChildOfClass("Humanoid")
                and myChar:FindFirstChild("HumanoidRootPart") then

                local myHum = myChar:FindFirstChildOfClass("Humanoid")
                local targetHRP = targetChar.HumanoidRootPart
                local myHRP = myChar.HumanoidRootPart

                if IsSpeedMatching and tHum then
                    myHum.WalkSpeed = tHum.WalkSpeed
                end

                if MovementMode == "Mirror" then
                    local offset = targetHRP.CFrame.LookVector * -2

                    myHRP.CFrame = targetHRP.CFrame + offset
                    myHRP.AssemblyLinearVelocity = targetHRP.AssemblyLinearVelocity
                    myHRP.AssemblyAngularVelocity = targetHRP.AssemblyAngularVelocity

                elseif MovementMode == "Normal" then
                    local _, targetY, _ = targetHRP.CFrame:ToOrientation()

                    myHRP.CFrame =
                        CFrame.new(myHRP.Position)
                        * CFrame.Angles(0, targetY, 0)

                    myHum:MoveTo(
                        targetHRP.Position
                        - (targetHRP.CFrame.LookVector * 2)
                    )
                end
            end
        else
            StopFollowing()
        end
    end)
end

local function ProcessCommands(text)
    local cmd = text:lower():gsub("%s+", "")

    if cmd == ";charme" or cmd == ";charma" or cmd == ";char" then
        if TargetPlayer then
            ChangeCharacterAppearance(TargetPlayer)
        end

        return true

    elseif cmd == ";unchar" then
        ResetCharacterAppearance()
        return true

    elseif cmd == ";follow" then
        FollowBtn:Activate()
        return true

    elseif cmd == ";speed" then
        SpeedBtn:Activate()
        return true

    elseif cmd == ";tool" then
        ToolBtn:Activate()
        return true
    end

    return false
end

-- إرسال الرسائل
-- =========================================================

function SendCustomMessage(textToSend)
    local text = textToSend or TextBox.Text

    if text == "" or text:match("^%s*$") then
        return
    end

    if ProcessCommands(text) then
        if not textToSend then
            TextBox.Text = ""
        end

        return
    end

    -- هنا يتم عكس الكلام فعلياً
    text = FilterAndReplaceWords(text)

    LastSentText = text

    if not textToSend then
        TextBox.Text = ""
    end

    local formattedText = text

    if UseCustomColor then
        local hexColor = SelectedColor:ToHex()

        formattedText = string.format(
            "<font color='#%s'>%s</font>",
            hexColor,
            text
        )
    end

    task.spawn(function()
        for _, obj in ipairs(ReplicatedStorage:GetDescendants()) do
            if obj:IsA("RemoteEvent") then
                local name = obj.Name:lower()

                if name:find("chat")
                    or name:find("message")
                    or name:find("say")
                    or name:find("send") then

                    pcall(function()
                        obj:FireServer(formattedText)
                    end)
                end
            end
        end
    end)
end

                or descendant:IsA("TextBox") then

                -- تجاهل نصوص الأدوات
                if IsInsideTool(descendant) then
                    return
                end

                ProcessTextLabel(descendant)

                if TextPropertyConnection then
                    TextPropertyConnection:Disconnect()
                end

                TextPropertyConnection =
                    descendant:GetPropertyChangedSignal("Text"):Connect(function()
                        if IsInsideTool(descendant) then
                            return
                        end

                        ProcessTextLabel(descendant)
                    end)
            end
        end)
    end

    if player.Character then
        HookCharacter(player.Character)
    end

    table.insert(
        GlobalChatConnections,
        player.CharacterAdded:Connect(function(character)
            task.wait(0.1)

            if TargetPlayer == player then
                HookCharacter(character)
            end
        end)
    )
end

local SelectingPlayer = false

TargetBtn.MouseButton1Click:Connect(function()
    if TargetPlayer then
        TargetPlayer = nil

        ClearChatHooks()
        StopFollowing()
        StopToolMatching()

        IsFollowing = false
        IsSpeedMatching = false
        IsToolMatching = false

        FollowBtn.Text = "تتبع الحركة والقفز (مُعطّل)"
        FollowBtn.BackgroundColor3 = Color3.fromRGB(40, 40, 40)

        SpeedBtn.Text = "مطابقة نفس سرعة اللاعب (مُعطّل)"
        SpeedBtn.BackgroundColor3 = Color3.fromRGB(40, 40, 40)

        ToolBtn.Text = "تقليد نمط مسك الأشياء باليد (مُعطّل)"
        ToolBtn.BackgroundColor3 = Color3.fromRGB(40, 40, 40)

        TargetBtn.Text = "اضغط لتحديد لاعب لتقليد كلامه (لا يوجد)"
        TargetBtn.BackgroundColor3 = Color3.fromRGB(40, 40, 40)

    else
        SelectingPlayer = true

        TargetBtn.Text = "انقر على شخصية اللاعب لتحديده..."
        TargetBtn.BackgroundColor3 = Color3.fromRGB(180, 120, 0)
    end
end)

Mouse.Button1Down:Connect(function()
    if not SelectingPlayer then
        return
    end

    local target = Mouse.Target

    if target then
        local model = target:FindFirstAncestorOfClass("Model")

        if model then
            local plr = Players:GetPlayerFromCharacter(model)

            if plr and plr ~= LocalPlayer then
                TargetPlayer = plr
                SelectingPlayer = false

                TargetBtn.Text =
                    "المستهدف: "
                    .. plr.DisplayName
                    .. " (@"
                    .. plr.Name
                    .. ")"

                TargetBtn.BackgroundColor3 = Color3.fromRGB(0, 120, 200)

                HookPlayerChat(plr)

                if IsFollowing then
                    StartFollowing()
                end

                if IsToolMatching then
                    StartToolMatching()
                end
            end
        end
    end
end)
